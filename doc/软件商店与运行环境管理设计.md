# DevEnv 软件商店与运行环境管理设计方案

**版本：** V1.0  
**日期：** 2026-10-14  
**目标：** 替换单一 PHP 环境页面，构建统一的软件商店与运行环境热拔插管理平台

---

## 一、核心设计理念

### 1.1 从"PHP 环境"到"软件商店 + 运行环境"

**现状问题：**
- 当前仅有单独的 "PHP 环境" 页面，功能单一
- Nginx、MySQL、Redis 等环境缺乏统一管理入口
- 用户需要手动登记每个环境，体验分散

**新方案：**
```
删除：【PHP 环境】单一页面
新增：【软件商店】（发现与安装）
新增：【运行环境】（已安装环境的统一管理）
```

### 1.2 设计原则

1. **统一管理**：所有运行环境（PHP、Nginx、MySQL、Redis 等）统一入口
2. **热拔插**：类似 VS Code 插件，一键安装/卸载，无需重启应用
3. **版本共存**：同一软件的多版本可并存，灵活切换
4. **集中存储**：所有运行环境统一安装到 DevEnv 目录下，便于管理和备份
5. **在线安装**：仅支持从软件商店在线安装，不支持导入外部环境
6. **状态可视**：清晰展示安装状态、版本信息、健康状态

## 三、数据模型设计

### 3.1 复用现有表结构

**`system_runtime_versions` 表（已存在，字段调整）：**
```sql
CREATE TABLE system_runtime_versions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    runtime_type TEXT NOT NULL,           -- php | nginx | mysql | redis ...
    version TEXT NOT NULL,                -- 8.3.12, 1.26.0, 8.4.0 ...
    variant TEXT,                         -- nts | ts | null (Nginx/MySQL 不需要)
    architecture TEXT NOT NULL,           -- x64 | x86 | arm64
    install_path TEXT NOT NULL,           -- 相对于 DevEnv 安装目录的路径
    status TEXT NOT NULL DEFAULT 'available',  -- available | installing | installed | failed | removed
    metadata_json TEXT NOT NULL DEFAULT '{}',  -- 扩展信息（JSON）
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    UNIQUE (runtime_type, version, variant, architecture)
);

-- 索引优化
CREATE INDEX IF NOT EXISTS idx_system_runtime_versions_type_status 
    ON system_runtime_versions(runtime_type, status);
```

**字段说明：**
- `install_path`: 存储相对路径（如 `runtimes\php\8.3.12-nts-x64`），便于 DevEnv 整体迁移
- `variant`: PHP 特有字段，区分 NTS (Non-Thread-Safe) 和 TS (Thread-Safe) 版本
- `status`: 新增 `removed` 状态，标识已卸载但保留配置的环境

**`metadata_json` 字段扩展用途：**
```json
{
```

```go
    // 3. 创建任务记录（安装任务）
    taskID := s.taskRepo.Create(ctx, "install", packageKey, version)
    
    // 4. 下载文件到临时目录 {DevEnvRoot}/temp/downloads/
    tempFile := s.downloadWithProgress(ctx, taskID, downloadURL)
    
    // 5. 验证文件完整性（SHA256）
    if !s.verifyChecksum(tempFile, expectedSHA256) {
        return errors.New("文件校验失败")
    }
    
    // 6. 解压到目标目录
    if err := s.extract(tempFile, installPath); err != nil {
        return err
    }
    
    // 7. 验证可执行文件
    executable := filepath.Join(installPath, s.getExecutableName(packageKey))
    if !fileExists(executable) {
        return errors.New("可执行文件不存在")
    }
    
    // 8. 注册到 runtime_versions（install_path 存储相对路径）
    relativePath := s.getRelativePath(installPath)
    runtimeID := s.runtimeRepo.Create(ctx, runtime.RuntimeVersion{
        RuntimeType:  runtime.RuntimeType(packageKey),
        Version:      version,
        Variant:      variant,
        Architecture: architecture,
        InstallPath:  relativePath,
        Status:       runtime.StatusInstalled,
    })
    
    // 9. 清理临时文件
    os.Remove(tempFile)
    
    // 10. 发布安装完成事件
    s.publisher.Emit("runtime:installed", runtimeID)
    
    return nil
}

func (s *StoreService) buildInstallPath(packageKey, version, variant, arch string) string {
    // 构建标准目录名
    dirName := version
    if variant != "" {
        dirName += "-" + variant
    }
    dirName += "-" + arch
    
    return filepath.Join(s.devEnvRoot, "runtimes", packageKey, dirName)
}
```

```go
// ListAllRuntimes 获取所有运行环境（按类型分组）
func (s *RuntimeService) ListAllRuntimes(ctx context.Context) (map[runtime.RuntimeType][]runtime.RuntimeVersion, error) {
    all, err := s.runtimeRepo.ListAll(ctx)
    if err != nil {
        return nil, err

// ScanRuntimesDirectory 扫描 DevEnv runtimes 目录
func (s *RuntimeService) ScanRuntimesDirectory(ctx context.Context) error {
    // 1. 读取 {DevEnvRoot}/runtimes/ 目录
    runtimesDir := filepath.Join(s.devEnvRoot, "runtimes")
    
    // 2. 遍历各运行时类型目录（php, nginx, mysql, redis, postgresql）
    types := []string{"php", "nginx", "mysql", "redis", "postgresql"}
    for _, runtimeType := range types {
        typeDir := filepath.Join(runtimesDir, runtimeType)
        if !dirExists(typeDir) {
            continue
        }
        
        // 3. 遍历版本目录（如 8.3.12-nts-x64）
        entries, _ := os.ReadDir(typeDir)
        for _, entry := range entries {
            if !entry.IsDir() {
                continue
            }
            
            // 4. 解析目录名，提取版本、variant、架构
            version, variant, arch := s.parseVersionDirName(entry.Name())
            
            // 5. 检查数据库是否已存在记录
            existing, _ := s.runtimeRepo.GetByTypeVersionVariantArch(ctx, runtimeType, version, variant, arch)
            if existing != nil {
                // 已存在，验证健康状态
                s.HealthCheck(ctx, existing.ID)
                continue
            }
            
            // 6. 验证可执行文件
            installPath := filepath.Join(typeDir, entry.Name())
            executable := filepath.Join(installPath, s.getExecutableName(runtimeType))
            if !fileExists(executable) {
                continue
            }
            
            // 7. 自动注册到数据库
            relativePath := s.getRelativePath(installPath)
            s.runtimeRepo.Create(ctx, runtime.RuntimeVersion{
                RuntimeType:  runtime.RuntimeType(runtimeType),
                Version:      version,
                Variant:      variant,
                Architecture: arch,
                InstallPath:  relativePath,
                Status:       runtime.StatusInstalled,
                MetadataJSON: `{"installedBy": "scanned"}`,
            })
        }
    }
    
    return nil
}

func (s *RuntimeService) parseVersionDirName(dirName string) (version, variant, arch string) {
    // 解析目录名格式：8.3.12-nts-x64 或 1.26.0-x64
    parts := strings.Split(dirName, "-")
    if len(parts) == 3 {
        return parts[0], parts[1], parts[2]
    } else if len(parts) == 2 {
        return parts[0], "", parts[1]
    }
    return dirName, "", "x64"
}

// UninstallRuntime 卸载运行环境
func (s *RuntimeService) UninstallRuntime(ctx context.Context, runtimeID int64, deleteFiles bool) error {
    // 1. 检查是否被服务实例引用
    instances, _ := s.serviceRepo.ListByRuntimeID(ctx, runtimeID)
    if len(instances) > 0 {
        runningCount := 0
        for _, inst := range instances {
            status := s.processMgr.GetStatus(ctx, inst.ID)
            if status.Status == service.StatusRunning {
                runningCount++
            }
        }
        if runningCount > 0 {
            return appErrors.New(appErrors.CodeResourceInUse, 
                fmt.Sprintf("该运行环境正被 %d 个运行中的服务使用，请先停止服务", runningCount))
        }
        return appErrors.New(appErrors.CodeResourceInUse, 
            fmt.Sprintf("该运行环境被 %d 个服务引用，请先删除相关服务", len(instances)))
    }

    // 2. 检查是否被项目引用
    projects, _ := s.projectRepo.ListByRuntimeID(ctx, runtimeID)
    if len(projects) > 0 {
        return appErrors.New(appErrors.CodeResourceInUse, 
            fmt.Sprintf("该运行环境被 %d 个项目引用，请先切换项目版本", len(projects)))
    }

    // 3. 获取运行环境信息
    rt, err := s.runtimeRepo.GetByID(ctx, runtimeID)
    if err != nil || rt == nil {
        return appErrors.New(appErrors.CodeNotFound, "运行环境不存在")
    }

    // 4. 删除数据库记录
    if err := s.runtimeRepo.Delete(ctx, runtimeID); err != nil {
        return err
    }

    // 5. 如果 deleteFiles=true，删除物理文件
    if deleteFiles {
        installPath := filepath.Join(s.devEnvRoot, rt.InstallPath)
        if err := os.RemoveAll(installPath); err != nil {
            // 记录错误但不回滚数据库操作
            s.logger.Error("删除运行环境文件失败", "path", installPath, "error", err)
        }
    }

    // 6. 发布卸载事件
    s.publisher.Emit("runtime:uninstalled", runtimeID)

    return nil
}
```

```typescript
async function handleQuickSwitchVersion(projectId: number, newRuntimeId: number) {
  try {
    // 1. 调用切换接口
    await switchProjectPHPVersion(projectId, newRuntimeId)
    
    // 2. 刷新项目信息
    await loadProjectDetail(projectId)
    
    // 3. 提示成功
    toast.add({
      severity: 'success',
      summary: '版本切换成功',
      detail: '项目 PHP 版本已更新，相关服务实例已自动切换',
      life: 3000
    })
  } catch (err: any) {
    // 处理兼容性警告
    if (err.code === 'VERSION_COMPATIBILITY_WARNING') {
      showCompatibilityDialog(err.details)
    } else {
      toast.add({
        severity: 'error',
        summary: '切换失败',
        detail: err.message,
        life: 3500
      })
    }
  }
}
```

## 九、总结

本设计方案通过**软件商店 + 运行环境**的双页面架构，替换原有单一的 PHP 环境页面，实现：

✅ **统一管理**：所有运行环境集中管理，体验一致  
✅ **热拔插**：一键安装/卸载，无需重启应用  
✅ **多版本共存**：灵活切换，满足不同项目需求  
✅ **自动发现**：智能扫描本地环境，减少手动配置  
✅ **可扩展性**：预留插件化架构，支持社区贡献

该方案复用现有 `system_runtime_versions` 表结构，无需大规模数据库迁移，分阶段实施，风险可控。
